cmake_minimum_required(VERSION 3.24)
project(SatchelNativeRewriter LANGUAGES C CXX)
include(FetchContent)
set(CAPSTONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_CSTOOL OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
set(CAPSTONE_ARM64_SUPPORT ON CACHE BOOL "" FORCE)
FetchContent_Declare(capstone URL https://github.com/capstone-engine/capstone/archive/d2e5739cbb43df93d589a3a726c8f878b95beccb.tar.gz URL_HASH SHA256=cc7a1542ded0424e47ef2ae275ca8050ec2ebfe870255ebd4375147f17ee88b0)
FetchContent_MakeAvailable(capstone)
option(NR_BUILD_SHARED "Build the rewriter as a shared library for the Mac test harness" ON)
option(NR_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(NR_BUILD_SHARED)
    add_library(native_rewriter SHARED src/rewriter.cpp)
else()
    add_library(native_rewriter STATIC src/rewriter.cpp)
endif()
if(NR_SANITIZE)
    target_compile_options(native_rewriter PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(native_rewriter PRIVATE -fsanitize=address,undefined)
endif()
target_compile_features(native_rewriter PRIVATE cxx_std_17)
target_include_directories(native_rewriter PUBLIC include PRIVATE ${capstone_SOURCE_DIR}/include)
target_link_libraries(native_rewriter PRIVATE capstone)
target_compile_options(native_rewriter PRIVATE -Wall -Wextra -Werror)
set_target_properties(native_rewriter PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

include(CTest)
if(BUILD_TESTING)
    add_executable(encoder_tests tests/encoders.cpp)
    target_compile_features(encoder_tests PRIVATE cxx_std_17)
    target_include_directories(encoder_tests PRIVATE include ${capstone_SOURCE_DIR}/include)
    target_link_libraries(encoder_tests PRIVATE capstone)
    if(NR_SANITIZE)
        target_compile_options(encoder_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(encoder_tests PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME encoder_boundaries COMMAND encoder_tests)
endif()
